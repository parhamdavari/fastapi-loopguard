```python
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse
import asyncio
from concurrent.futures import ThreadPoolExecutor
from typing import Any

import helpers  # noqa: F401  (the provided API)

app = FastAPI()

# Module-level executor for offloading CPU-bound image resizing
_executor: ThreadPoolExecutor | None = None


def _get_executor() -> ThreadPoolExecutor:
    global _executor
    if _executor is None:
        _executor = ThreadPoolExecutor(max_workers=4)
    return _executor


@app.post("/thumbnail")
async def thumbnail(request: Request) -> Any:
    data: bytes = await request.body()
    loop = asyncio.get_event_loop()

    # Run CPU-bound resize_image in a thread to avoid blocking the event loop
    thumbnail_bytes: bytes = await loop.run_in_executor(
        _get_executor(), helpers.resize_image, data
    )

    return JSONResponse(content={"size": len(thumbnail_bytes)}, status_code=200)
```
