```python
from fastapi import FastAPI
import helpers
import asyncio

app = FastAPI()

@app.get("/portfolio")
async def portfolio(symbols: str):
    # Parse comma-separated symbols, stripping whitespace
    symbol_list = [s.strip() for s in symbols.split(",")]
    
    # Create concurrent tasks for async price fetching
    tasks = [helpers.afetch_price(sym) for sym in symbol_list]
    results = await asyncio.gather(*tasks)
    
    # Build dictionary mapping symbol to price
    prices = {sym: price for sym, price in zip(symbol_list, results)}
    
    return {"prices": prices}
```
