```python
from fastapi import FastAPI, Request, status
from fastapi.responses import JSONResponse

import helpers  # noqa: F401

app = FastAPI()


@app.post("/documents")
async def save_document_endpoint(request: Request):
    data = await request.json()
    name = data.get("name")
    text = data.get("text")
    # Ideally: validate input, but instructions say nothing about it.
    num_bytes = helpers.save_document(name, text)
    return JSONResponse(content={"bytes": num_bytes}, status_code=status.HTTP_201_CREATED)
```
