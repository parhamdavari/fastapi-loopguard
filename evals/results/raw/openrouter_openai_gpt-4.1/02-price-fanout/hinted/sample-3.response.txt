```python
from fastapi import FastAPI, Query
from fastapi.responses import JSONResponse
import asyncio

import helpers  # noqa: F401

app = FastAPI()


@app.get("/portfolio")
async def get_portfolio(symbols: str = Query(...)):
    symbol_list = [s for s in symbols.split(",") if s]
    # Run fetches in parallel (concurrently) using async helpers
    prices_list = await asyncio.gather(*(helpers.afetch_price(symbol) for symbol in symbol_list))
    prices_dict = dict(zip(symbol_list, prices_list))
    return JSONResponse(content={"prices": prices_dict}, status_code=200)
```
