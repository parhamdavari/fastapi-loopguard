```python
from fastapi import FastAPI, Depends

import helpers

app = FastAPI()

billing = helpers.BillingClient()


def get_billing() -> helpers.BillingClient:
    """FastAPI dependency providing the billing client."""
    return billing


@app.post("/charges", status_code=201)
def create_charge(amount: float, billing_client: helpers.BillingClient = Depends(get_billing)):
    receipt_id = billing_client.charge(amount)
    return {"receipt": receipt_id}
```
