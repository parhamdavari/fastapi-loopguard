```python
from fastapi import FastAPI
import helpers
import asyncio

app = FastAPI()


@app.get("/portfolio")
async def portfolio(symbols: str = ""):
    # Split the comma‑separated list, ignore empty entries
    sym_list = [s.strip() for s in symbols.split(",") if s.strip()]
    if not sym_list:
        return {"prices": {}}

    # Launch all async fetches concurrently
    tasks = [helpers.afetch_price(sym) for sym in sym_list]
    prices = await asyncio.gather(*tasks)

    # Build the response dictionary
    result = {sym: price for sym, price in zip(sym_list, prices)}
    return {"prices": result}
```
